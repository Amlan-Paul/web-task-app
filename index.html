সরাসরি শিট দেখুন

গুগল শিট API বা Apps Script ওয়েব অ্যাপ সংযোগ

এই ওয়েব অ্যাপটি আপনার গুগল শিটের সাথে ২ ভাবে কাজ করতে পারে:
১. লোকাল ক্যাশ ও অফলাইন মোড (ডিফল্ট: শিটের প্রি-লোডেড ডেটা দিয়ে তৎক্ষণাৎ ব্যবহারযোগ্য)।
২. Google Apps Script Web App (দ্বিমুখী লাইভ রিড/রাইট ডাটাবেস)।

Apps Script ডিপ্লয়মেন্ট ইউআরএল দিলে নতুন টাস্ক বা আপডেট সরাসরি গুগল শিটে সেভ হবে।

১ মিনিটে Google Apps Script ব্যাকএন্ড সেটআপ কোড

  1. আপনার Google Sheet-এ যান এবং মেনু থেকে Extensions > Apps Script-এ ক্লিক করুন।
  2. নিচের কোডটি কপি করে এডিটরে পেস্ট করুন এবং Save করুন।
  3. উপরে ডানদিকের Deploy > New deployment বাটনে চাপুন।
  4. Select type হিসেবে Web app নির্বাচন করুন, Access দিন Anyone এবং Deploy করুন।
  5. পাওয়া Web App URL কপি করে উপরের বক্সে পেস্ট করে "সংরক্ষণ করুন" চাপুন!
// Google Apps Script (Code.gs) for WebTask Database
function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var taskSheet = ss.getSheetByName("Tasks_Database");
  var updateSheet = ss.getSheetByName("Website_Updates_Log");
  
  var taskData = taskSheet ? taskSheet.getDataRange().getValues() : [];
  var updateData = updateSheet ? updateSheet.getDataRange().getValues() : [];
  
  var result = {
    status: "success",
    tasks: formatRows(taskData),
    updates: formatRows(updateData)
  };
  return ContentService.createTextOutput(JSON.stringify(result))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var params = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    
    if (params.type === "ADD_TASK") {
      var sheet = ss.getSheetByName("Tasks_Database");
      sheet.appendRow([
        params.task.id,
        params.task.title,
        params.task.page,
        params.task.category,
        params.task.status,
        params.task.priority,
        params.task.assignedTo,
        params.task.version,
        params.task.dueDate,
        Utilities.formatDate(new Date(), "GMT+6", "yyyy-MM-dd HH:mm"),
        params.task.notes
      ]);
    } else if (params.type === "ADD_UPDATE") {
      var sheet = ss.getSheetByName("Website_Updates_Log");
      sheet.appendRow([
        params.update.id,
        params.update.version,
        params.update.title,
        params.update.page,
        params.update.date,
        params.update.status,
        params.update.summary
      ]);
    }
    return ContentService.createTextOutput(JSON.stringify({status: "success"}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({status: "error", message: err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function formatRows(data) {
  if (data.length <= 1) return [];
  var headers = data[0];
  var rows = [];
  for (var i = 1; i < data.length; i++) {
    var item = {};
    for (var j = 0; j < headers.length; j++) {
      item[headers[j]] = data[i][j];
    }
    rows.push(item);
  }
  return rows;
}
